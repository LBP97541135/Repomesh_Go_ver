// Cross-batch recovery extends the existing web/src/modelRecovery.ts module. It shares only

export type OperationLocator = Readonly<
  | { kind: "provider_save"; actor: ActorID; id: SaveID }
  | { kind: "model_test"; actor: ActorID; id: TestID }
  | { kind: "model_apply"; actor: ActorID; projectId: ProjectID; id: ApplicationID }
  | { kind: "issue_create"; actor: ActorID; projectId: ProjectID; id: CreationID }

export type SecretFreeSaveSnapshot = SaveFields & Readonly<{ secretMode: "keep" | "replace" }>;
// Only these non-secret snapshots may persist in sessionStorage. localStorage holds only a
// RecoveryIndex. Storage keys include actor and operation identity before lookup.
export type SessionRecovery = Readonly<
  | { locator: ProviderSaveLocator; snapshot: SecretFreeSaveSnapshot }
  | { locator: ModelTestLocator; body: TestBody }
  | { locator: ModelApplyLocator; body: { previewId: PreviewID } }
  | { locator: IssueCreateLocator; body: IssueCreateBody }
>;
export type SessionRecoveryFor<L extends OperationLocator> = Extract<SessionRecovery, { locator: L }>;
export type ReadTicket<L extends OperationLocator = OperationLocator> = Readonly<{ locator: L; generation: number; requestSerial: number }>;
export type OperationState<L extends OperationLocator, T> = Readonly<

  | { state: "removed"; locator: L; generation: number }
  | { state: "reauthenticate" | "restricted"; locator: L; generation: number }
>;

// Reducers ignore mismatched actor/locator/generation; at most one test read is in flight.
// 404/503/abort retain locator and frozen input. 410 clears payload but does not restore key reuse.
// Save body Key released after sending; save recovery can only GET/close, never rebuild original Key.
// Durable rejected/closed permits an explicit new save; Issue has no durable rejected receipt.
// Unknown Issue retains frozen body; user cannot silently replace expected context under old id.
// An exact retry, where its business contract permits one, reuses the stored body and original id.
// Shared storage never manufactures input and never decides whether an operation may retry.
// Authentication recovery passes an allowlisted Destination; browser never auto-resubmits on login.
// B06 renders stable creation receipt. Detail/rooms routes stay deferred, not fabricated as ready.

// Provider-save storage remains byte-compatible with the implemented B04 keys:
// localStorage repomesh.model.save-index.v1 and sessionStorage
// repomesh.model.save.v1:{actor}:{id}. Added kinds use distinct kind-specific prefixes through
// the same storage helper. Identity loss clears session payloads but retains the minimal indexes.
// Indexes never grant read access and are filtered by the authenticated actor before display.
// The current SaveLocator export migrates to ProviderSaveLocator. ModelSavePage migrates its
// locator, ticket, state, and readSessionRecovery call. ModelSettingsPage keeps the existing
// provider-save persistence path. session.ts replaces clearModelRecovery with
// clearRecoveryPayloads and keeps clearAllOperationInputs for B03. model.test.mjs keeps the B04
// key, actor-isolation, secret-removal, and stale-generation assertions, then adds each new
// locator and reducer. The old SaveLocator type is deleted after these callers compile together.

// web/src/projectRecovery.ts remains the B03 project_create/project_update owner. It keeps
// OperationIdentity, key normalization, input parsing and freezing, prepare/load, project routes,
// its reducer, and repomesh.project.operation*. It may use shared read tickets later, but this
// B05/B06 design does not move project input storage or retry rules into modelRecovery.ts.

// Existing api.ts Result<T> carries diagnostics/retryAt. WriteContext is obtained from the
